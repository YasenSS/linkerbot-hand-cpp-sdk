# Read-only adapter entry point. No arguments/help perform zero driver I/O.
add_executable(windows_readonly windows_readonly.cpp)
target_link_libraries(windows_readonly PRIVATE linkerbot::cpp-sdk)
if(MSVC)
    target_compile_options(windows_readonly PRIVATE /utf-8)
endif()
if(LINKERBOT_BUILD_TESTS)
    add_test(NAME windows_readonly_help COMMAND windows_readonly)
    add_test(NAME windows_readonly_invalid COMMAND windows_readonly --motion)
    set_tests_properties(windows_readonly_invalid PROPERTIES WILL_FAIL TRUE)
    add_test(NAME windows_readonly_slcan_fd COMMAND windows_readonly --list --backend slcan --channel COM10 --mode fd)
    add_test(NAME windows_readonly_slcan_allowlist COMMAND windows_readonly --list --backend slcan)
    add_test(NAME windows_readonly_slcan_uart COMMAND windows_readonly --list --backend slcan --channel COM10 --serial-baud 0)
    add_test(NAME windows_readonly_slcan_fd_rate COMMAND windows_readonly --list --backend slcan --channel COM10 --mode fd --slcan-protocol canable-fd --data-bitrate 4000000)
    add_test(NAME windows_readonly_slcan_profile COMMAND windows_readonly --list --backend slcan --channel COM10 --slcan-protocol unknown)
    add_test(NAME windows_readonly_slcan_startup COMMAND windows_readonly --list --backend slcan --channel COM10 --serial-startup-ms 5001)
    set_tests_properties(windows_readonly_slcan_fd windows_readonly_slcan_allowlist windows_readonly_slcan_uart windows_readonly_slcan_fd_rate windows_readonly_slcan_profile windows_readonly_slcan_startup PROPERTIES WILL_FAIL TRUE)
endif()
install(TARGETS windows_readonly RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
